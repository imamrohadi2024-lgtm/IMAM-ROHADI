function renderBarChart(perBulan){
var container=document.getElementById('barChartContainer');
if(!perBulan||perBulan.length===0){container.innerHTML='<div class="flex h-full items-center justify-center text-sm text-slate-400">Tidak ada data</div>';return;}
var bulanNama=['Jan','Feb','Mar','Apr','Mei','Jun','Jul','Agu','Sep','Okt','Nov','Des'];
var maxCount=Math.max.apply(null,perBulan.map(function(b){return b.count;}))||1;
var W=600,H=260,padL=40,padR=20,padT=20,padB=40;
var chartW=W-padL-padR,chartH=H-padT-padB;
var barW=chartW/12*0.6,gap=chartW/12*0.4;
var isDark=document.documentElement.classList.contains('dark');
var gridColor=isDark?'#374151':'#e2e8f0';
var textColor=isDark?'#94a3b8':'#64748b';
var barColor1=isDark?'#60a5fa':'#2d5a87';
var barColor2=isDark?'#3b82f6':'#4a90e2';
var svg='<svg viewBox="0 0 '+W+' '+H+'" class="h-full w-full" preserveAspectRatio="xMidYMid meet">';
for(var i=0;i<=4;i++){
var y=padT+chartH*(i/4);
var val=Math.round(maxCount*(1-i/4));
svg+='<line x1="'+padL+'" y1="'+y+'" x2="'+(W-padR)+'" y2="'+y+'" stroke="'+gridColor+'" stroke-width="1" stroke-dasharray="3,3"/>';
svg+='<text x="'+(padL-8)+'" y="'+(y+4)+'" text-anchor="end" fill="'+textColor+'" font-size="10" font-weight="600">'+val+'</text>';
}
perBulan.forEach(function(b,idx){
var x=padL+idx*(chartW/12)+gap/2;
var h=(b.count/maxCount)*chartH;
var y=padT+chartH-h;
svg+='<g class="chart-bar">';
svg+='<rect class="chart-bar-rect" x="'+x+'" y="'+y+'" width="'+barW+'" height="'+h+'" rx="4" fill="url(#barGrad)" style="height:0;y:'+(padT+chartH)+'" data-target-y="'+y+'" data-target-h="'+h+'"/>';
svg+='<text x="'+(x+barW/2)+'" y="'+(H-padB+18)+'" text-anchor="middle" fill="'+textColor+'" font-size="10" font-weight="600">'+bulanNama[idx]+'</text>';
if(b.count>0){svg+='<text x="'+(x+barW/2)+'" y="'+(y-6)+'" text-anchor="middle" fill="'+textColor+'" font-size="10" font-weight="700">'+b.count+'</text>';}
svg+='</g>';
});
svg+='<defs><linearGradient id="barGrad" x1="0" y1="0" x2="0" y2="1"><stop offset="0%" stop-color="'+barColor2+'"/><stop offset="100%" stop-color="'+barColor1+'"/></linearGradient></defs>';
svg+='</svg>';
container.innerHTML=svg;
setTimeout(function(){
var rects=container.querySelectorAll('.chart-bar-rect');
for(var i=0;i<rects.length;i++){var r=rects[i];r.style.height=r.getAttribute('data-target-h')+'px';r.style.y=r.getAttribute('data-target-y');}
},50);
}

function renderPieChart(perJenis){
var container=document.getElementById('pieChartContainer');
if(!perJenis){container.innerHTML='<div class="flex h-64 items-center justify-center text-sm text-slate-400">Tidak ada data</div>';return;}
var keys=Object.keys(perJenis);
if(keys.length===0){container.innerHTML='<div class="flex h-64 items-center justify-center text-sm text-slate-400">Belum ada data</div>';return;}
var total=keys.reduce(function(s,k){return s+perJenis[k];},0);
var colors=['#1e3a5f','#28a745','#f39c12','#dc3545','#6f42c1','#17a2b8','#e83e8c','#20c997'];
var isDark=document.documentElement.classList.contains('dark');
if(isDark)colors=['#60a5fa','#34d399','#fbbf24','#f87171','#a78bfa','#22d3ee','#f472b6','#2dd4bf'];
var size=220,r=85,ir=50,cx=size/2,cy=size/2;
var svg='<svg viewBox="0 0 '+size+' '+size+'" width="'+size+'" height="'+size+'">';
var startAngle=-Math.PI/2;
keys.forEach(function(k,i){
var val=perJenis[k];
var angle=(val/total)*Math.PI*2;
var endAngle=startAngle+angle;
var largeArc=angle>Math.PI?1:0;
var x1=cx+r*Math.cos(startAngle),y1=cy+r*Math.sin(startAngle);
var x2=cx+r*Math.cos(endAngle),y2=cy+r*Math.sin(endAngle);
var ix1=cx+ir*Math.cos(startAngle),iy1=cy+ir*Math.sin(startAngle);
var ix2=cx+ir*Math.cos(endAngle),iy2=cy+ir*Math.sin(endAngle);
var d='M '+x1+' '+y1+' A '+r+' '+r+' 0 '+largeArc+' 1 '+x2+' '+y2+' L '+ix2+' '+iy2+' A '+ir+' '+ir+' 0 '+largeArc+' 0 '+ix1+' '+iy1+' Z';
var pct=Math.round((val/total)*100);
svg+='<path class="chart-pie-segment" d="'+d+'" fill="'+colors[i%colors.length]+'" data-label="'+escapeHtml(k)+'" data-count="'+val+'" data-pct="'+pct+'"/>';
startAngle=endAngle;
});
svg+='<circle cx="'+cx+'" cy="'+cy+'" r="'+(ir-4)+'" fill="'+(isDark?'#1e293b':'#ffffff')+'"/>';
svg+='<text x="'+cx+'" y="'+(cy-4)+'" text-anchor="middle" fill="'+(isDark?'#f1f5f9':'#1e3a5f')+'" font-size="20" font-weight="800">'+total+'</text>';
svg+='<text x="'+cx+'" y="'+(cy+14)+'" text-anchor="middle" fill="'+(isDark?'#94a3b8':'#64748b')+'" font-size="10" font-weight="600">Total</text>';
svg+='</svg>';
var legend='<div class="mt-4 w-full space-y-1.5">';
keys.forEach(function(k,i){
var val=perJenis[k];
var pct=Math.round((val/total)*100);
legend+='<div class="flex items-center gap-2 text-xs"><span class="h-3 w-3 shrink-0 rounded" style="background:'+colors[i%colors.length]+'"></span><span class="flex-1 truncate text-slate-600">'+escapeHtml(k)+'</span><span class="font-bold text-slate-700">'+val+' <span class="text-slate-400 font-normal">('+pct+'%)</span></span></div>';
});
legend+='</div>';
container.innerHTML='<div class="flex flex-col items-center">'+svg+legend+'</div>';
}

function hitungLamaCuti(){
var mulai = document.getElementById('formTglMulai').value;
var selesai = document.getElementById('formTglSelesai').value;
var el = document.getElementById('formLamaCuti');
if (mulai && selesai) {
var t1 = parseISOLokal(mulai);
var t2 = parseISOLokal(selesai);
if (!t1 || !t2) { el.value = '0 hari kerja'; return; }
if (t2 >= t1) {
var jml = 0;
var d = new Date(t1.getFullYear(), t1.getMonth(), t1.getDate());
var end = new Date(t2.getFullYear(), t2.getMonth(), t2.getDate());
var libur = {};
HARI_LIBUR_FE.forEach(function(t){ libur[t] = true; });
while (d <= end) {
var h = d.getDay();
var k = d.getFullYear() + '-' + String(d.getMonth()+1).padStart(2,'0') + '-' + String(d.getDate()).padStart(2,'0');
if (h !== 0 && h !== 6 && !libur[k]) jml++;
d.setDate(d.getDate() + 1);
}
el.value = jml + ' hari kerja';
} else { el.value = '0 hari kerja'; }
} else { el.value = '0 hari kerja'; }
}

function submitPengajuanLengkap(){
var formData={
nip:document.getElementById('formNip').value.trim(),
nama:document.getElementById('formNama').value.trim(),
statusKepegawaian:document.getElementById('formStatusKepegawaian').value,
jabatan:document.getElementById('formJabatan').value.trim(),
unitKerja:document.getElementById('formUnitKerja').value,
nipAtasanLangsung:document.getElementById('formNipAtasan').value.trim(),
atasanLangsung:document.getElementById('formAtasanNama').value.trim(),
noHP:document.getElementById('formNoHP').value.trim(),
email:document.getElementById('formEmail').value.trim(),
jenisCuti:document.getElementById('formJenisCuti').value,
tanggalMulai:document.getElementById('formTglMulai').value,
tanggalSelesai:document.getElementById('formTglSelesai').value,
alamat:document.getElementById('formAlamat').value.trim(),
alasan:document.getElementById('formAlasan').value.trim()
};
var normalizeNip=function(v){return String(v||'').replace(/\s+/g,'');};
if(!formData.nip){ showToast('NIP wajib diisi!','warning'); return; }
if(!formData.nama){ showToast('Nama pemohon belum terisi, silakan klik tombol cari 🔍 NIP!','warning'); return; }
if(!formData.unitKerja){ showToast('Unit kerja belum terisi!','warning'); return; }
if(!formData.nipAtasanLangsung){ showToast('NIP Atasan wajib diisi!','warning'); return; }
if(normalizeNip(formData.nip)===normalizeNip(formData.nipAtasanLangsung)){ showToast('NIP Atasan tidak boleh sama dengan NIP pemohon!','warning'); return; }
if(!formData.atasanLangsung){ showToast('Nama Atasan belum ditemukan, klik cari 🔍 NIP atasan!','warning'); return; }

var wajib=['jenisCuti','tanggalMulai','tanggalSelesai','alamat','alasan','noHP'];
for(var i=0;i<wajib.length;i++){ if(!formData[wajib[i]]){ showToast('Semua kolom bertanda * harus diisi!','warning'); return; } }

var t1 = parseISOLokal(formData.tanggalMulai);
var t2 = parseISOLokal(formData.tanggalSelesai);
if(!t1 || !t2){ showToast('Format tanggal tidak valid!','warning'); return; }
if(t2 < t1){ showToast('Tanggal selesai cuti tidak boleh mendahului tanggal mulai!','warning'); return; }

if(formData.jenisCuti==='Cuti Tahunan' && kuotaForm){
if(kuotaForm.diminta<=0){ showToast('Durasi 0 hari kerja efektif! Periksa kembali tanggal cuti.','warning'); return; }
if(kuotaForm.diminta>kuotaForm.sisa){ showToast('Pengajuan melebihi sisa cuti! Sisa: '+kuotaForm.sisa+' hari kerja','error'); return; }
}

var jenisDok=document.getElementById('docJenis').value;
if(jenisDok==='Lainnya'){ jenisDok=document.getElementById('docJenisLainnya').value.trim(); if(!jenisDok){ showToast('Isi nama jenis dokumen manual!','warning'); return; } }
if(selectedFile && !jenisDok){ showToast('Pilih jenis dokumen untuk file yang diunggah!','warning'); return; }

var btn=document.getElementById('btnKirimPengajuan'),orig=btn.innerHTML;
btn.disabled=true;
btn.innerHTML='<span class="spinner" style="width:14px;height:14px;border-width:2px;border-color:#fff;border-top-color:transparent;"></span> Mengirim ke database...';
showLoading('Mengirim pengajuan cuti...');

google.script.run.withSuccessHandler(function(resCuti){
if(!resCuti.success){
hideLoading(); btn.disabled=false; btn.innerHTML=orig;
showToast((resCuti.message||'Gagal mengajukan'),'error');
return;
}
var noPengajuan=resCuti.noPengajuan;
var tglPengajuan=resCuti.tanggalPengajuan || '-';

if(selectedFile){
var reader=new FileReader();
reader.onload=function(e){
var fd={
nip:formData.nip, nama:formData.nama, unitKerja:formData.unitKerja,
jenisDokumen:jenisDok, noPengajuan:noPengajuan,
keterangan:document.getElementById('docKeterangan').value.trim(),
fileData:e.target.result.split(',')[1], fileName:selectedFile.name,
fileType:selectedFile.type, fileSize:selectedFile.size
};
google.script.run.withSuccessHandler(function(resDok){
hideLoading(); btn.disabled=false; btn.innerHTML=orig;
showSukses('Pengajuan cuti berhasil tercatat di database!\nNo: '+noPengajuan+'\nTanggal Pengajuan: '+tglPengajuan+(resDok.success?'\nDokumen pendukung terunggah.':''));
resetFormCuti();
setTimeout(function(){ loadAllData(); },500);
}).withFailureHandler(function(){
hideLoading(); btn.disabled=false; btn.innerHTML=orig;
showSukses('Pengajuan berhasil!\nNo. Pengajuan: '+noPengajuan+'\nTanggal Pengajuan: '+tglPengajuan+'\nDokumen gagal diunggah.');
resetFormCuti();
}).uploadDokumen(fd);
};
reader.readAsDataURL(selectedFile);
} else {
hideLoading(); btn.disabled=false; btn.innerHTML=orig;
showSukses('Pengajuan cuti berhasil tercatat di database!\nNo. Pengajuan: '+noPengajuan+'\nTanggal Pengajuan: '+tglPengajuan);
resetFormCuti();
setTimeout(function(){ loadAllData(); },500);
}
}).withFailureHandler(function(err){
hideLoading(); btn.disabled=false; btn.innerHTML=orig;
showToast('Error sistem: ' + ((err && err.message) || err),'error');
}).submitCuti(formData);
}

function prosesResetPassword(){
var pl=document.getElementById('resetPasswordLama').value,
pb=document.getElementById('resetPasswordBaru').value,
pk=document.getElementById('resetPasswordKonfirmasi').value;
var msgEl=document.getElementById('resetPasswordMsg');
if(!pl||!pb||!pk){msgEl.innerHTML='<div class="alert alert-warning"><span class="material-symbols-rounded mr-1 align-middle">warning</span>Semua wajib diisi</div>';return;}
if(pb.length<6){msgEl.innerHTML='<div class="alert alert-warning"><span class="material-symbols-rounded mr-1 align-middle">warning</span>Min 6 karakter</div>';return;}
if(pb===pl){msgEl.innerHTML='<div class="alert alert-warning"><span class="material-symbols-rounded mr-1 align-middle">warning</span>Tidak boleh sama</div>';return;}
if(pb!==pk){msgEl.innerHTML='<div class="alert alert-warning"><span class="material-symbols-rounded mr-1 align-middle">warning</span>Konfirmasi tidak cocok</div>';return;}
document.getElementById('btnResetSubmit').disabled=true;
google.script.run.withSuccessHandler(function(res){
document.getElementById('btnResetSubmit').disabled=false;
if(res.success){showToast('Password diganti!','success');closeModalResetPassword();}
else msgEl.innerHTML='<div class="alert alert-danger"><span class="material-symbols-rounded mr-1 align-middle">error</span>'+escapeHtml(res.message)+'</div>';
}).withFailureHandler(function(){document.getElementById('btnResetSubmit').disabled=false;msgEl.innerHTML='<div class="alert alert-danger"><span class="material-symbols-rounded mr-1 align-middle">error</span>Error</div>';}).resetPassword(currentUser.username,pl,pb);
}
